# Python dependencies for CCS_Artifacts (TSDP-LST, VGG-GTSRB).
#
# Two runtimes are needed because the GPU side and the TEE/CPU side need
# different PyTorch wheels (CUDA vs CPU). Recommended setup:
#
#   # GPU side (artifact2_e2e/backbone.py + artifact1_nas/search.py)
#   python3 -m venv my_venv_cuda && source my_venv_cuda/bin/activate
#   pip install --upgrade pip
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu121
#   pip install -r requirements.txt
#
#   # TEE / CPU side (artifact2_e2e/tee_server.py, runs under gramine-sgx)
#   python3 -m venv my_venv && source my_venv/bin/activate
#   pip install --upgrade pip
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements.txt
#
# Tested with Python 3.10 on Ubuntu 22.04 + Intel SGX2 (Gramine 1.8).
